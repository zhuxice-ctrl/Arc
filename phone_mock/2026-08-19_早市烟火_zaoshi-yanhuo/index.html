摊位列表

GET /api/stalls
请求参数
{
  "status": "open",  // open | closing | closed | all
  "category": "noodle"
}
响应示例
[
  {
    "id": "congbaohui",
    "name": "老张葱包桧",
    "category": "noodle",
    "openTime": "06:00",
    "closeTime": "10:00",
    "location": "凤起路巷口",
    "signature": "葱包桧+甜面酱",
    "heat": 4,
    "ownerQuote": "三十年只做一锅葱包桧",
    "distance": "320m"
  }
]

请求体
{
  "stallId": "congbaohui",

}

{
  "success": true,
  "checkinId": "ck_20240819_001",
  "totalCheckins": 4,
  "streakDays": 7
