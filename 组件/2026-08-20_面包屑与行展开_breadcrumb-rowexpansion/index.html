面包屑 Breadcrumb

HTML 骨架

<nav aria-label="当前位置" id="myBreadcrumb"></nav>

JS 初始化

Breadcrumb.init(document.getElementById('myBreadcrumb'), {
  items: [
    { id: 'home', label: '首页', href: '/' },
    { id: 'samples', label: '样品中心', href: '/samples' },
    { id: 'batch', label: '2026-B07 批次', href: '/b/07', current: true }
  ],
  maxVisible: 5,        // 超过此项数则折叠中间项为省略号
  onNavigate: (item) => {
    console.log('跳转到', item.label);
    // return false 可阻止默认跳转
  },
  onLoadChildren: null, // function(parentId) => Promise
                        // 返回新 items 数组，触发 loading 态
});

表格行展开 Row Expansion

HTML 骨架

<table class="re-table" id="myTable">
  <thead class="re-table__head">
    <tr>
      <th scope="col" style="width:44px"></th>
      <th scope="col">列一</th>
      <th scope="col">列二</th>
    </tr>
  </thead>
  <tbody class="re-table__body" id="myTableBody">
    <!-- 行由 JS 注入 -->
  </tbody>
</table>

JS 初始化

RowExpansion.init(document.getElementById('myTableBody'), {
  rows: [
    {
      id: 'row-1',
      cells: ['值一', '值二', '值三'],
      detail: {
        // 同步详情：直接给数据
        fields: [
          { label: 'pH 值', value: '7.24', mono: true },
          { label: '备注', value: '...', note: true }
        ]
      }
    }
  ],
  // 或使用异步加载：返回 Promise<detail>
  asyncLoader: (rowId) => fetch(`/api/rows/${rowId}`).then(r => r.json()),
  colSpan: 5,       // 详情行跨列数，通常 = 列数
  disabled: false,  // 或 (row) => boolean
  onExpand: (rowId, expanded) => {}
});
