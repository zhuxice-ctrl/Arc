用法

引入组件 CSS 与 JS 后，通过 new Segmented(options) 创建实例。

const seg = new Segmented({
  container: '#target',   // 选择器或 DOM 元素
  layout: 'equal',        // equal 等宽 | fit 自适应
  items: [
    { label: '日视图', value: 'day' },
    { label: '周视图', value: 'week', badge: 3 },
    { label: '月视图', value: 'month', disabled: true },
  ],
  defaultValue: 'day',
  onChange: (value) => { console.log(value); },
});

// API
seg.getValue();
seg.setValue('week');
seg.setDisabled(true);
seg.setLoading(true);
seg.setBadge('week', 5);
seg.recalcIndicator();
seg.destroy();
